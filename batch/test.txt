echo "Processing $(wc -l < $CHUNKNAME) files for chunk $PROCESS"

#######################################################################
# Stage input CAFs locally (workers can't read /pnfs/ via POSIX)
#######################################################################
mkdir -p data
LOCAL_CHUNK=local_chunk.txt
> $LOCAL_CHUNK

while IFS= read -r p; do
    b=$(basename "$p")
    echo "Staging: $p"
    ifdh cp --cp_maxretries=2 --web_timeout=300 "$p" "data/$b"
    echo "data/$b" >> $LOCAL_CHUNK
done < $CHUNKNAME

echo "Staged $(wc -l < $LOCAL_CHUNK) files locally"

#######################################################################
# Run UpdateReweight on local files
#######################################################################
printf -v OUTNAME "chunk_%04d_reweighted.caf.root" "$PROCESS"

UpdateReweight \
    -c $MYWD/zexpansion_weighter.ParameterHeader.fcl \
    -i $LOCAL_CHUNK \
    -o $OUTNAME
